# M2.2 — Nesting / Sheet-Layout DXF Export Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Furniture→Nesting "Export DXF (Nest)" button that exports `activeAssembly.nest` as a ZIP of DXF files — one true-millimetre cut-layout DXF per sheet (stock boundary + panel rectangles + labels).

**Architecture:** A pure renderer `renderNestSheetDxf(sheet, meta)` emits one DXF per `NestSheet` onto three layers (SHEET / OUTLINE / INFO) using the existing `createDxfDocument` writer at 1:1 mm. A pure helper `buildNestZipEntries(nest, materials)` maps each sheet to a zip entry; `ExportNestDxfCommand` zips them via `buildDxfZip` and downloads (mirroring `ExportPanelManufacturingDxfCommand`). A ribbon button wires it up.

**Tech Stack:** TypeScript, React, vitest (jsdom), fflate (zip), hand-rolled AC1015 DXF writer.

## Global Constraints

- **Per-task gate (M2.1 lesson #1):** after EVERY task run `pnpm typecheck` from `packages/app/` (or repo root) and confirm 0 errors before committing. A struct-vs-string type bug rode 3 tasks in M2.1 with no per-task gate.
- **Unit tests:** vitest, run from `packages/app/`: `pnpm test:unit` (or `npx vitest run <path>`). Test files are `*.spec.ts` under `src/**/__tests__/`. Config: `packages/app/vitest.config.ts` (jsdom, globals, `./src/test-setup.ts`).
- **Cross-package type imports:** `NestState`, `NestSheet`, `PlacedPanel`, `MaterialRegistry`, `MaterialEntry`, `MaterialId` — **copy the exact import specifier verbatim** from the sibling files (`packages/app/src/commands/ExportNestPdfCommand.ts` imports `NestState` + `MaterialRegistry`; `ExportNestPngCommand.ts` imports `NestSheet`). Do NOT invent an alias.
- **Rotated-footprint convention (M2.1 lesson #2 — VERIFIED against `packGuillotine`):** `PlacedPanel.length` is ALWAYS the X-extent and `PlacedPanel.width` ALWAYS the Y-extent at `(x, y)`. `rotated` is purely informational — the footprint swap is *already baked into* `length`/`width`. The renderer draws `width = panel.length`, `height = panel.width` with **NO** rotated-based swap. This matches `SheetSvg.tsx` (`<rect width={p.length} height={p.width}/>`). The spec §1 prose "drawn width = rotated ? panel.width : panel.length" is WRONG — do not implement it.
- **Layers are entity group-8 names only.** `dxfWriter`'s LAYER table is a fixed list that does NOT include SHEET/OUTLINE/INFO; entities just carry the layer string in group 8 and viewers auto-create it (exactly as M2.1's manufacturing DXF does). No `dxfWriter.ts` change.
- **Coordinates:** 1:1 mm, Y-up, anchored at the panel's own `(x, y)` (no Y-flip — the spec anchors panel rects at `(panel.x, panel.y)` directly). `fmt()` formats every coord as `toFixed(4)` (e.g. `300` → `300.0000`).

function lwpolys(dxf: string): Array<{ layer: string; minX: number; minY: number; w: number; h: number }> {
  const lines = dxf.split('\n');
  const out: Array<{ layer: string; minX: number; minY: number; w: number; h: number }> = [];
  for (let i = 0; i < lines.length; i++) {
    if (lines[i] === '  0' && lines[i + 1] === 'LWPOLYLINE') {
      let layer = '';
      const xs: number[] = [];
      const ys: number[] = [];
      for (let j = i + 2; j < lines.length; j++) {
        if (lines[j] === '  0') break;
        if (lines[j] === '  8') layer = lines[j + 1]!;
        if (lines[j] === ' 10') { xs.push(parseFloat(lines[j + 1]!)); ys.push(parseFloat(lines[j + 3]!)); }

```ts
import { createDxfDocument } from './dxfWriter';
import { MANUFACTURING_LAYERS } from './manufacturingLayers';
import type { NestSheet } from '<copy specifier from ExportNestPngCommand.ts>';

/**
 * One true-mm cut-layout DXF for a single nest sheet.
 * Layers: SHEET (stock boundary), OUTLINE (panel cut rects), INFO (labels).
 * `rotated` is already baked into length/width — no swap here (see plan Global Constraints).
 */
export function renderNestSheetDxf(
  sheet: NestSheet,
  meta: { sheetCount: number; materialName: string },
): string {
  const doc = createDxfDocument();
  const L = sheet.sheetLengthMm;
  const W = sheet.sheetWidthMm;

  doc.lwpolyline(MANUFACTURING_LAYERS.sheet, [
    { x: 0, y: 0 }, { x: L, y: 0 }, { x: L, y: W }, { x: 0, y: W },
  ], true);

  for (const p of sheet.panels) {
    const w = p.length; // X-extent (rotation already baked in)
    const h = p.width;  // Y-extent
    doc.lwpolyline(MANUFACTURING_LAYERS.outline, [
      { x: p.x, y: p.y }, { x: p.x + w, y: p.y },
      { x: p.x + w, y: p.y + h }, { x: p.x, y: p.y + h },
    ], true);
    doc.text(MANUFACTURING_LAYERS.info, p.x + w / 2, p.y + h / 2, 15, `#${p.itemNumber}  ${p.length}×${p.width}`);
  }

  doc.text(
    MANUFACTURING_LAYERS.info,
    0, W + 30, 30,
    `Sheet ${sheet.sheetIndex + 1}/${meta.sheetCount} · ${meta.materialName} · util ${sheet.utilizationPct}%`,
  );

  return doc.toString();
}
```

import { renderNestSheetDxf } from '../drawing/renderNestSheetDxf';
import type { NestState } from '<copy specifier from ExportNestPdfCommand.ts>';
import type { MaterialRegistry } from '<copy specifier from ExportNestPdfCommand.ts>';

function sanitize(name: string): string {
  return name.replace(/[^A-Za-z0-9_-]/g, '_');
}

export function buildNestZipEntries(
  nest: NestState,
  materials: MaterialRegistry,
): Record<string, string> {
  const entries: Record<string, string> = {};
  for (const sheet of nest.sheets) {
    const materialName = materials.find((m) => m.id === sheet.materialId)?.name ?? sheet.materialId;
    const sheetCount = nest.sheets.filter((s) => s.materialId === sheet.materialId).length;
    entries[`${sanitize(sheet.materialId)}-sheet-${sheet.sheetIndex + 1}.dxf`] =
      renderNestSheetDxf(sheet, { sheetCount, materialName });
  }
  return entries;
}

```ts
import { buildDxfZip } from '../drawing/dxfZip';

declare global {
  // eslint-disable-next-line no-var
  var __zilventorNestDxfBytes: Uint8Array | undefined;
}

interface FsHandle {
  createWritable: () => Promise<{ write: (data: Uint8Array) => Promise<void>; close: () => Promise<void> }>;
}
type SaveFilePicker = (opts: { suggestedName: string; types: Array<{ description: string; accept: Record<string, string[]> }> }) => Promise<FsHandle>;

export class ExportNestDxfCommand {
  constructor(private opts: { nest: NestState; materials: MaterialRegistry; suggestedName: string }) {}

  async apply(): Promise<void> {
    const bytes = buildDxfZip(buildNestZipEntries(this.opts.nest, this.opts.materials));
    globalThis.__zilventorNestDxfBytes = bytes;
    const picker = (globalThis as unknown as { showSaveFilePicker?: SaveFilePicker }).showSaveFilePicker;
    if (typeof picker === 'function') {
      try {
        const handle = await picker({
          suggestedName: this.opts.suggestedName,
          types: [{ description: 'ZIP', accept: { 'application/zip': ['.zip'] } }],
        });
        const w = await handle.createWritable();
        await w.write(bytes);
        await w.close();
      } catch (err) {
        const e = err as { name?: string };
        if (e?.name === 'AbortError') return;
        if (typeof document === 'undefined') return;
        this.downloadFallback(bytes);
      }
      return;
    }
    if (typeof document !== 'undefined') this.downloadFallback(bytes);
  }

  private downloadFallback(bytes: Uint8Array): void {
    const blob = new Blob([bytes as unknown as Uint8Array<ArrayBuffer>], { type: 'application/zip' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = this.opts.suggestedName;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }
}
```
