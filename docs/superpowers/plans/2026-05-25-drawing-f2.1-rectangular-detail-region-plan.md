# Drawing F2.1 — Rectangular Detail Region Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Detail views can magnify an axis-aligned rectangular sub-region of a parent drawing view, alongside the existing circle, via the same pure-2D clip→translate→scale pipeline (no kernel call).

**Architecture:** `RegionParent` becomes a `circle | rect` discriminated union (manifest v18→v19 backfills `kind:'circle'`). A new Liang–Barsky `clipEdgesToRect` mirrors `clipEdgesToCircle`; `projectDetailView` dispatches the clip step on `regionParent.kind` (translate+scale stay shared). A `detailRect` ToolMode (corner-to-corner marquee) and a "Detail (Rect)" ribbon button create rects; F2.4's `EditDetailRegionCommand` + drag handles extend to rect (body=move, 4 corners=resize). Rect branches land in BOTH renderer paths (live React + string-SVG/PDF export).

**Tech Stack:** TypeScript (pnpm monorepo: `@zilventor/shared`, `@zilventor/app`), Zustand store + command bus, React/SVG live canvas, string-SVG + jsPDF export, Vitest (unit/e2e/perf), Playwright (visual verification). `exactOptionalPropertyTypes: true` enforced.

**Batches (implement in order — later batches depend on earlier interface contracts):**
1. Types union + migration v18→v19 + fingerprint union (`@zilventor/shared`)
2. Rect clip math (Liang–Barsky) + `projectDetailView` dispatch (`@zilventor/shared`)
3. Commands + `detailRect`/`editDetailRegion`-rect ToolMode + `MIN_DETAIL_RECT_MM` (`@zilventor/app`)
4. Live React UI: ribbon button, overlay/border/edit-handles rect branches, canvas marquee wiring (`@zilventor/app`)
5. Export renderers (SVG/PDF/PNG) rect branches + E2E (T-6-16) + perf + manual-smoke doc (`@zilventor/app`)

### Canonical cross-batch contracts (every batch MUST match these exactly)

```ts
// packages/shared/src/drawing/types.ts — Batch 1 defines
export type RegionParent =
  | { readonly kind: 'circle'; readonly center: Vec2; readonly radiusMm: number }
  | { readonly kind: 'rect';   readonly center: Vec2; readonly widthMm: number; readonly heightMm: number };
```

- `fingerprintDetailView(parentInputFingerprint, regionParent: RegionParent, scale)` — union-aware (Batch 1).
- `clipEdgesToRect(edges, rect: {center; widthMm; heightMm})` + `projectDetailView` dispatch on `regionParent.kind` (Batch 2).
- `AddDetailViewCommand` constructor takes a unified `regionParent: RegionParent` field (Batch 3); sets BOTH `activeDocument` AND `activeDrawing`.
- `EditDetailRegionCommand` + `reprojectDetailRegion` carry `RegionParent` as `prevRegion`/`newRegion` (Batch 3).
- `useAppStore.updateDetailRegionLive(sheetId, detailViewId, region: RegionParent)` — widened to the union (Batch 3).
- ToolMode (Batch 3): `'detailRect'` mode actions `enterDetailRectMode` / `detailRectPickParent` / `detailRectPickFirstCorner` / `detailRectPickSecondCorner` / `detailRectPickDrop` / `detailRectCancel`, stages `idle → pickFirstCorner → pickSecondCorner → pickDrop`. `'editDetailRegion'` gains a `'draggingCorner'` stage; edit drag returns `RegionParent`.
- `MIN_DETAIL_RECT_MM = 1.0` in `packages/app/src/drawing/drawingConstants.ts` (Batch 3).
- Rect corner-resize math: dragged corner `C'`, opposite corner `O` fixed → `center=midpoint(C',O)`, `widthMm=max(|C'.x−O.x|, MIN_DETAIL_RECT_MM)`, `heightMm=max(|C'.y−O.y|, MIN_DETAIL_RECT_MM)`.
- Reuse `data-role="detail-border"` and `data-role="detail-callout-label"` on rect elements (no new roles); rect branches land in BOTH live React and export paths (Batches 4 + 5).

1. **Types** — `RegionParent` becomes a discriminated union (`kind:'circle'|'rect'`).

```ts
/**
 * Synchronous djb2 fingerprint for a DetailView.
 * Accepts the RegionParent discriminated union (F2.1).
 * Payload layout:
 *   circle: parentFp|kind|center.x|center.y|radiusMm|num|den
 *   rect:   parentFp|kind|center.x|center.y|widthMm|heightMm|num|den
 * Floats rounded to 1e-3 mm via .toFixed(3) for determinism.
 * Including `kind` in the payload ensures a circle and a rect with the
 * same bounding box produce distinct fingerprints.
 */
export function fingerprintDetailView(
  parentInputFingerprint: string,
  regionParent: RegionParent,
  scale: { numerator: number; denominator: number },
): string {
  let payload: string;
  if (regionParent.kind === 'circle') {
    payload = [
      parentInputFingerprint,
      'circle',
      regionParent.center.x.toFixed(3),
      regionParent.center.y.toFixed(3),
      regionParent.radiusMm.toFixed(3),
      scale.numerator,
      scale.denominator,
    ].join('|');
  } else {
    payload = [
      parentInputFingerprint,
      'rect',
      regionParent.center.x.toFixed(3),
      regionParent.center.y.toFixed(3),
      regionParent.widthMm.toFixed(3),
      regionParent.heightMm.toFixed(3),
      scale.numerator,
      scale.denominator,
    ].join('|');
  }
  return djb2(payload);
}
```

```ts
// packages/shared/src/drawing/clipEdgesToRect.ts
//
// Liang–Barsky clip of line segments against an axis-aligned rectangle.
// Mirrors clipEdgesToCircle.ts in structure.

import type { Edge2D } from './projectedGeometry';
import type { Segment } from './clipEdgesToCircle';

export type { Segment };

type Rect = { center: { x: number; y: number }; widthMm: number; heightMm: number };

/**
 * Clip segment [p0, p1] against axis-aligned box
 * [cx − w/2, cx + w/2] × [cy − h/2, cy + h/2]
 * using the Liang–Barsky algorithm.
 * Returns [] (clipped away) or one surviving sub-segment.
 */
export function clipLineToRect(
  p0: readonly [number, number],
  p1: readonly [number, number],
  rect: Rect,
): Segment[] {
  const { center, widthMm, heightMm } = rect;
  const xMin = center.x - widthMm / 2;
  const xMax = center.x + widthMm / 2;
  const yMin = center.y - heightMm / 2;
  const yMax = center.y + heightMm / 2;

  const dx = p1[0] - p0[0];
  const dy = p1[1] - p0[1];

  // Liang–Barsky: maintain [tEnter, tExit] = [0, 1] and clip each boundary.
  // For each boundary: p*t + q >= 0  must hold for the segment to be inside.
  // p[k] = direction component (may be zero for axis-aligned segments).
  // q[k] = signed distance from start point to the boundary.
  // Boundaries: left (xMin), right (xMax), bottom (yMin), top (yMax).
  const p: readonly number[] = [-dx, dx, -dy, dy];
  const q: readonly number[] = [
    p0[0] - xMin,   // left:   p0.x - xMin >= 0 means p0 is right of left edge
    xMax - p0[0],   // right:  xMax - p0.x >= 0 means p0 is left of right edge
    p0[1] - yMin,   // bottom: p0.y - yMin >= 0 means p0 is above bottom edge
    yMax - p0[1],   // top:    yMax - p0.y >= 0 means p0 is below top edge
  ];

  let tEnter = 0.0;
  let tExit  = 1.0;

  for (let i = 0; i < 4; i++) {
    const pi = p[i]!;
    const qi = q[i]!;

    if (Math.abs(pi) < 1e-12) {
      // Segment is parallel to this boundary.
      // If q[i] < 0 the segment is entirely outside — reject.
      if (qi < 0) return [];
      // Otherwise parallel and inside this boundary — no clip needed.
    } else {
      const t = qi / pi;
      if (pi < 0) {
        // p[i] < 0: segment travels from inside to outside → t is an entry bound.
        if (t > tEnter) tEnter = t;
      } else {
        // p[i] > 0: segment travels from outside to inside → t is an exit bound.
        if (t < tExit) tExit = t;
      }
    }
  }

  if (tEnter > tExit + 1e-9) return []; // missed the box

  const cp0: [number, number] = [p0[0] + tEnter * dx, p0[1] + tEnter * dy];
  const cp1: [number, number] = [p0[0] + tExit  * dx, p0[1] + tExit  * dy];
  return [{ p0: cp0, p1: cp1 }];
}

/**
 * Clip a polyline (array of [x,y] points) against an axis-aligned rect.
 * Returns zero or more Segments — one per consecutive pair that survives.
 */
export function clipPolylineToRect(
  points: ReadonlyArray<readonly [number, number]>,
  rect: Rect,
): Segment[] {
  if (points.length < 2) return [];
  const result: Segment[] = [];
  for (let i = 0; i < points.length - 1; i++) {
    const segs = clipLineToRect(points[i]!, points[i + 1]!, rect);
    result.push(...segs);
  }
  return result;
}

/**
 * Clip an array of Edge2D against an axis-aligned rect in parent-local mm space.
 * Mirrors clipEdgesToCircle: preserves id/kind, conditional-spreads sourceFaceIds
 * (exactOptionalPropertyTypes), suffixes survivors _clip0/_clip1/…
 */
export function clipEdgesToRect(
  edges: readonly Edge2D[],
  rect: Rect,
): Edge2D[] {
  const result: Edge2D[] = [];
  for (const edge of edges) {
    const segments = clipPolylineToRect(edge.points, rect);
    for (let i = 0; i < segments.length; i++) {
      const clipped: Edge2D = {
        id: `${edge.id}_clip${i}`,
        kind: edge.kind,
        points: [segments[i]!.p0, segments[i]!.p1],
        ...(edge.sourceFaceIds !== undefined && { sourceFaceIds: edge.sourceFaceIds }),
      };
      result.push(clipped);
    }
  }
  return result;
}
```

```ts
    readonly projection: ProjectedGeometry;
  };
  readonly regionParent: RegionParent;
  readonly detailScale: { readonly numerator: number; readonly denominator: number };
  readonly showHidden: boolean;
};

export type DetailViewOutput = {
  readonly visibleEdges: readonly Edge2D[];
  readonly hiddenEdges: readonly Edge2D[];
  readonly bbox: { minX: number; minY: number; maxX: number; maxY: number };
};

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

function translateAndScale(
  edges: Edge2D[],
  cx: number,
  cy: number,
  factor: number,
): Edge2D[] {
  return edges.map(edge => ({
    ...edge,
    points: edge.points.map(
      ([x, y]): readonly [number, number] => [
        (x - cx) * factor,
        (y - cy) * factor,
      ],
    ),
  }));
}

function computeBbox(
  edges: readonly Edge2D[],
): { minX: number; minY: number; maxX: number; maxY: number } {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  for (const edge of edges) {
    for (const [x, y] of edge.points) {
      if (x < minX) minX = x;
      if (y < minY) minY = y;
      if (x > maxX) maxX = x;
      if (y > maxY) maxY = y;
    }
  }
  if (!isFinite(minX)) return { minX: 0, minY: 0, maxX: 0, maxY: 0 };
  return { minX, minY, maxX, maxY };
}

/**
 * Clip edges against the regionParent boundary, dispatching on kind.
 * circle → clipEdgesToCircle; rect → clipEdgesToRect.
 * The translate (center→origin) and scale steps are shared — both shapes are center-based.
 */
function clipEdges(
  edges: readonly Edge2D[],
  regionParent: RegionParent,
): Edge2D[] {
  if (regionParent.kind === 'circle') {
    return clipEdgesToCircle(edges, regionParent.center, regionParent.radiusMm);
  }
  // kind === 'rect'
  return clipEdgesToRect(edges, {
    center: regionParent.center,
    widthMm: regionParent.widthMm,
    heightMm: regionParent.heightMm,
  });
}

// ---------------------------------------------------------------------------
// Main entry (spec §6.1)
// ---------------------------------------------------------------------------

/**
 * Clips the parent view's projected edges against the region and transforms
 * the clipped edges to detail-local mm space.
 *
 * Step 1: Clip parent edges against regionParent (circle or rect).
 * Step 2: Translate clipped points by -regionParent.center (center → origin).
 * Step 3: Scale by detailScaleFactor =
 *         (detailScale.num/detailScale.den) / (parentScale.num/parentScale.den).
 * Step 4: Recompute bbox from transformed points.
 */
export function projectDetailView(input: DetailViewInput): DetailViewOutput {
  const { parentView, regionParent, detailScale, showHidden } = input;
  const { center } = regionParent;

  // Step 1: Clip
  const clippedVisible = clipEdges(parentView.projection.visibleEdges, regionParent);
  const clippedHidden = showHidden
    ? clipEdges(parentView.projection.hiddenEdges, regionParent)
    : [];

  // Step 2 + 3: Translate to origin then scale to detail-local mm
  const parentScale =
    parentView.scale.numerator / parentView.scale.denominator;
  const detailScaleVal =
    detailScale.numerator / detailScale.denominator;
  const factor = detailScaleVal / parentScale;

  const visibleEdges = translateAndScale(clippedVisible, center.x, center.y, factor);
  const hiddenEdges  = translateAndScale(clippedHidden,  center.x, center.y, factor);

  // Step 4: Recompute bbox
  const allEdges = [...visibleEdges, ...hiddenEdges];
  const bbox = computeBbox(allEdges);

  return { visibleEdges, hiddenEdges, bbox };
}
```

```ts
// packages/app/src/commands/EditDetailRegionCommand.ts
import type { Command } from './Command';
import type { DetailView, DrawingDocument, RegionParent } from '@zilventor/shared';
import { useAppStore } from '../store/useAppStore';
import { reprojectDetailRegion } from './reprojectDetailRegion';

export type EditDetailRegionOpts = {
  sheetId: string;
  detailViewId: string;
  prevRegion: RegionParent;
  newRegion:  RegionParent;
};

export class EditDetailRegionCommand implements Command {
  readonly name = 'EditDetailRegion';

  constructor(private readonly opts: EditDetailRegionOpts) {}

  async apply(): Promise<void> {
    this._write(this.opts.newRegion);
  }

  async revert(): Promise<void> {
    this._write(this.opts.prevRegion);
  }

  private _write(region: RegionParent): void {
    const state = useAppStore.getState();
    const doc = state.activeDocument;
    if (!doc || doc.kind !== 'drawing') return;

    const drawing = doc.document;
    const sheet = drawing.sheets.find(s => s.id === this.opts.sheetId);
    if (!sheet) return;

    const detail = sheet.views.find(v => v.id === this.opts.detailViewId);
    if (!detail || detail.kind !== 'detail') return;
    const detailView = detail as DetailView;

    const parentView = sheet.views.find(v => v.id === detailView.parentViewId);
    if (!parentView || !parentView.projection) return;

    const next = reprojectDetailRegion(
      detailView,
      parentView as typeof parentView & { projection: NonNullable<typeof parentView['projection']> },
      region,
    );

    const nextDoc: DrawingDocument = {
      ...drawing,
      sheets: drawing.sheets.map(s =>
        s.id === this.opts.sheetId
          ? { ...s, views: s.views.map(v => v.id === this.opts.detailViewId ? next : v) }
          : s
      ),
    };
    useAppStore.setState({ activeDocument: { kind: 'drawing', document: nextDoc }, activeDrawing: nextDoc });
  }
}
```

```ts
// packages/app/src/commands/AddDetailViewCommand.ts
import type { Command } from './Command';
import type { DrawingDocument, DetailView, SectionView, RegionParent } from '@zilventor/shared';
import { allocateCalloutLabel, projectDetailView, fingerprintDetailView } from '@zilventor/shared';
import { useAppStore } from '../store/useAppStore';

export type AddDetailViewOpts = {
  drawingId: string;
  sheetId: string;
  parentViewId: string;
  regionParent: RegionParent;   // circle or rect; built by ToolMode on commit
  dropAt: { x: number; y: number };  // sheet-space position mm
};

export class AddDetailViewCommand implements Command {
  readonly name = 'AddDetailView';
  private newViewId: string | null = null;

  constructor(private readonly opts: AddDetailViewOpts) {}

  async apply(): Promise<void> {
    const state = useAppStore.getState();
    const doc = state.activeDocument;
    if (!doc || doc.kind !== 'drawing') return;

    const drawing = doc.document;
    const sheet = drawing.sheets.find(s => s.id === this.opts.sheetId);
    if (!sheet) throw new Error(`sheet ${this.opts.sheetId} not found`);

    // 1. Resolve parent view.
    const parentView = sheet.views.find(v => v.id === this.opts.parentViewId);
    if (!parentView) throw new Error('parent-not-found');

    // 2. Require parent projection to be populated.
    if (!parentView.projection) throw new Error('parent-projection-missing');

    // 3. Allocate label from shared pool.
    const existingLabels = sheet.views
      .filter((v): v is SectionView | DetailView =>
        v.kind === 'section' || v.kind === 'detail')
      .map(v => v.label);
    const label = allocateCalloutLabel(existingLabels);

    // 4. Determine showHidden from parent.
    const showHidden =
      parentView.kind !== 'section' &&
      (parentView as { showHidden?: boolean }).showHidden === true;

    // 5. Default scale: 2:1.
    const defaultScale = { numerator: 2, denominator: 1 };

    // 6. Clip parent edges → projectDetailView (pure 2D, no kernel).
    const projection = projectDetailView({
      parentView: parentView as typeof parentView & { projection: NonNullable<typeof parentView.projection> },
      regionParent: this.opts.regionParent,
      detailScale: defaultScale,
      showHidden,
    });

    // 7. Compute fingerprint.
    const inputFingerprint = fingerprintDetailView(
      (parentView as { inputFingerprint?: string }).inputFingerprint ?? '',
      this.opts.regionParent,
      defaultScale,
    );

    // 8. Build DetailView.
    const newId = crypto.randomUUID();
    this.newViewId = newId;

    const view: DetailView = {
      id: newId,
      kind: 'detail',
      parentViewId: this.opts.parentViewId,
      regionParent: this.opts.regionParent,
      label,
      positionMm: this.opts.dropAt,
      scale: defaultScale,
      showHidden,
      projection,
      inputFingerprint,
    };

    // 9. Push onto sheet.views[].
    const nextDoc: DrawingDocument = {
      ...drawing,
      sheets: drawing.sheets.map(s =>
        s.id === this.opts.sheetId
          ? { ...s, views: [...s.views, view] }
          : s
      ),
    };
    // F6 gotcha: DrawingCanvas subscribes to activeDrawing — must set BOTH.
    useAppStore.setState({ activeDocument: { kind: 'drawing', document: nextDoc }, activeDrawing: nextDoc });
  }

  async revert(): Promise<void> {
    if (!this.newViewId) return;
    const state = useAppStore.getState();
    const doc = state.activeDocument;
    if (!doc || doc.kind !== 'drawing') return;

    const drawing = doc.document;
    const nextDoc: DrawingDocument = {
      ...drawing,
      sheets: drawing.sheets.map(s =>
        s.id === this.opts.sheetId
          ? { ...s, views: s.views.filter(v => v.id !== this.newViewId) }
          : s
      ),
    };
    useAppStore.setState({ activeDocument: { kind: 'drawing', document: nextDoc }, activeDrawing: nextDoc });
    this.newViewId = null;
  }
}
```

### Task 4.2 — `DetailRegionOverlay.tsx`: add rect callout branch

**File:** `packages/app/src/drawing/DetailRegionOverlay.tsx`

**Current state:** The component destructures `{ center, radiusMm }` directly from
`detailView.regionParent` (no discriminated-union branch). After Batch 1 the type is a
union, so this destructure will type-error. Replace with a switch on `kind`.

**Rect callout geometry (sheet space):**
- `parentScale = parentView.scale.numerator / parentView.scale.denominator`
- Rect in sheet space:
  - `x = parentView.positionMm.x + (center.x − widthMm/2) * parentScale`
  - `y = parentView.positionMm.y + (center.y − heightMm/2) * parentScale`
  - `width  = widthMm  * parentScale`
  - `height = heightMm * parentScale`
- Label position (upper-right of rect perimeter): same formula as circle but anchored
  to the rect's top-right corner plus a 5 mm diagonal offset:
  - `labelX = parentView.positionMm.x + (center.x + widthMm/2) * parentScale + 5 * cos45`
  - `labelY = parentView.positionMm.y + (center.y − heightMm/2) * parentScale − 5 * sin45`

**Rewritten component:**

```tsx
// packages/app/src/drawing/DetailRegionOverlay.tsx
import React from 'react';
import type { BaseView, ProjectedView, IsoView, SectionView, DetailView } from '@zilventor/shared';

interface Props {
  parentView: BaseView | ProjectedView | IsoView | SectionView;
  detailView: DetailView;
  /** Sheet zoom factor (default 1). Used to scale label font size. */
  zoom?: number;
}

const LABEL_FONT_SIZE_MM = 5;
const STROKE_WIDTH = 0.35;

export function DetailRegionOverlay({ parentView, detailView, zoom = 1 }: Props) {
  const parentScale = parentView.scale.numerator / parentView.scale.denominator;
  const rp = detailView.regionParent;
  const fontSize = LABEL_FONT_SIZE_MM / zoom;
  const cos45 = Math.SQRT2 / 2;
  const sin45 = Math.SQRT2 / 2;

  if (rp.kind === 'circle') {
    const cx = parentView.positionMm.x + rp.center.x * parentScale;
    const cy = parentView.positionMm.y + rp.center.y * parentScale;
    const r  = rp.radiusMm * parentScale;
    const labelX = cx + (r + 5) * cos45;
    const labelY = cy - (r + 5) * sin45;

    return (
      <g data-detail-callout-overlay={detailView.id}>
        <circle
          cx={cx}
          cy={cy}
          r={r}
          fill="none"
          stroke="black"
          strokeWidth={STROKE_WIDTH}
          data-role="detail-callout-label"
        />
        <text
          x={labelX}
          y={labelY}
          textAnchor="middle"
          dominantBaseline="central"
          fontSize={fontSize}
          fontFamily="sans-serif"
          data-role="detail-callout-label"
        >
          {detailView.label}
        </text>
      </g>
    );
  }

  // kind === 'rect'
  const { center, widthMm, heightMm } = rp;
  const rx = parentView.positionMm.x + (center.x - widthMm / 2) * parentScale;
  const ry = parentView.positionMm.y + (center.y - heightMm / 2) * parentScale;
  const rw = widthMm  * parentScale;
  const rh = heightMm * parentScale;
  // Label: upper-right corner of the rect + 5 mm diagonal offset
  const labelX = parentView.positionMm.x + (center.x + widthMm / 2) * parentScale + 5 * cos45;
  const labelY = parentView.positionMm.y + (center.y - heightMm / 2) * parentScale - 5 * sin45;

  return (
    <g data-detail-callout-overlay={detailView.id}>
      <rect
        x={rx}
        y={ry}
        width={rw}
        height={rh}
        fill="none"
        stroke="black"
        strokeWidth={STROKE_WIDTH}
        data-role="detail-callout-label"
      />
      <text
        x={labelX}
        y={labelY}
        textAnchor="middle"
        dominantBaseline="central"
        fontSize={fontSize}
        fontFamily="sans-serif"
        data-role="detail-callout-label"
      >
        {detailView.label}
      </text>
    </g>
  );
}
```

  - `sheetCy = parentView.positionMm.y + center.y * parentScale`
  - Half-extents: `hw = (widthMm / 2) * parentScale`, `hh = (heightMm / 2) * parentScale`
  - Rect in sheet space: `x = sheetCx - hw`, `y = sheetCy - hh`, `w = widthMm*parentScale`,
    `h = heightMm*parentScale`
- Body rect: transparent fill, dashed stroke (same style as circle body)
- 4 corner handles (`data-role="edit-corner-handle-NW"`, `…NE`, `…SE`, `…SW`):
  - NW: `(sheetCx - hw, sheetCy - hh)` — opposite corner is SE `(sheetCx + hw, sheetCy + hh)`
  - NE: `(sheetCx + hw, sheetCy - hh)` — opposite corner is SW `(sheetCx - hw, sheetCy + hh)`
  - SE: `(sheetCx + hw, sheetCy + hh)` — opposite corner is NW `(sheetCx - hw, sheetCy - hh)`
  - SW: `(sheetCx - hw, sheetCy + hh)` — opposite corner is NE `(sheetCx + hw, sheetCy - hh)`
  - Each handle cursor: `nw-resize` / `ne-resize` / `se-resize` / `sw-resize`
- Handle drag calls `useToolModeStore.getState().editDragStart(detailView.id, handleId, startRegion, anchorLocalPt)`
  where `handleId` is `'corner-NW'` / `'corner-NE'` / `'corner-SE'` / `'corner-SW'`
  (Batch 3 extended `editDragStart`'s second arg to cover these handle IDs).

```tsx
// packages/app/src/drawing/DetailRegionEditHandles.tsx
import React, { useCallback, useRef } from 'react';
import type { DetailView, DrawingView } from '@zilventor/shared';
import { useToolModeStore } from './toolMode';
import type { Region } from './toolMode';

const HANDLE_SIZE_MM = 3; // screen-constant handle side in sheet-mm at zoom=1; scales with 1/zoom

interface Props {
  parentView: DrawingView & { projection: NonNullable<DrawingView['projection']> };
  detailView: DetailView;
  zoom: number;
  onUpdateLive: (region: Region) => void;
  onCommit: (prevRegion: Region, newRegion: Region) => void;
  svgRef: React.RefObject<SVGSVGElement | null>;
  sheetW: number;
  sheetH: number;
}

function screenToSheet(
  clientX: number,
  clientY: number,
  svg: SVGSVGElement | null,
  sheetW: number,
  sheetH: number,
): { x: number; y: number } {
  const rect = svg?.getBoundingClientRect();
  const hasSize = rect && rect.width > 0 && rect.height > 0;
  return {
    x: hasSize ? ((clientX - rect.left) / (rect.width / sheetW)) : clientX,
    y: hasSize ? ((clientY - rect.top) / (rect.height / sheetH)) : clientY,
  };
}

function sheetToParentLocal(
  sheetPt: { x: number; y: number },
  parentView: DrawingView,
): { x: number; y: number } {
  const scale = parentView.scale.numerator / parentView.scale.denominator;
  return {
    x: (sheetPt.x - parentView.positionMm.x) / scale,
    y: (sheetPt.y - parentView.positionMm.y) / scale,
  };
}

// Shared drag wiring — attaches window pointermove/pointerup, RAF-throttled
function startDrag(
  svg: SVGSVGElement | null,
  sheetW: number,
  sheetH: number,
  parentView: DrawingView,
  rafPendingRef: React.MutableRefObject<boolean>,
  onMove: (localPt: { x: number; y: number }) => void,
  onUp: () => void,
): void {
  const handleMove = (ev: PointerEvent) => {
    if (rafPendingRef.current) return;
    rafPendingRef.current = true;
    requestAnimationFrame(() => {
      rafPendingRef.current = false;
      const sheetPt = screenToSheet(ev.clientX, ev.clientY, svg, sheetW, sheetH);
      const localPt = sheetToParentLocal(sheetPt, parentView);
      onMove(localPt);
    });
  };
  const handleUp = () => {
    onUp();
    window.removeEventListener('pointermove', handleMove);
    window.removeEventListener('pointerup', handleUp);
  };
  window.addEventListener('pointermove', handleMove);
  window.addEventListener('pointerup', handleUp);
}
```

```ts
function renderDetailCalloutGroups(views: readonly DrawingView[]): string {
  const groups: string[] = [];
  for (const view of views) {
    if (view.kind !== 'detail') continue;
    const dv = view as DetailView;
    const parent = views.find((v) => v.id === dv.parentViewId);
    if (!parent) continue;
    const parentScale = parent.scale.numerator / parent.scale.denominator;
    const cx = parent.positionMm.x + dv.regionParent.center.x * parentScale;
    const cy = parent.positionMm.y + dv.regionParent.center.y * parentScale;
    const rp = dv.regionParent;

    let shapeMarkup: string;
    let labelX: number;
    let labelY: number;

    if (rp.kind === 'rect') {
      const w = rp.widthMm * parentScale;
      const h = rp.heightMm * parentScale;
      const rx = cx - w / 2;
      const ry = cy - h / 2;
      shapeMarkup = `<rect x="${rx}" y="${ry}" width="${w}" height="${h}" fill="none" stroke="#000" stroke-width="0.35" data-role="detail-callout-label" data-detail-id="${escapeAttr(dv.id)}"/>`;
      // Label at top-right corner of the rect
      labelX = rx + w + 2;
      labelY = ry - 2;
    } else {
      // kind === 'circle'
      const r = rp.radiusMm * parentScale;
      const COS45 = Math.SQRT1_2;
      shapeMarkup = `<circle cx="${cx}" cy="${cy}" r="${r}" fill="none" stroke="#000" stroke-width="0.35" data-role="detail-callout-label" data-detail-id="${escapeAttr(dv.id)}"/>`;
      labelX = cx + (r + 5) * COS45;
      labelY = cy - (r + 5) * COS45;
    }

    groups.push(
      `  <g data-detail-callout="${escapeAttr(dv.id)}">` +
      `\n    ${shapeMarkup}` +
      `\n    <text x="${labelX}" y="${labelY}" font-size="5" font-family="sans-serif" text-anchor="start" data-role="detail-callout-label" data-detail-id="${escapeAttr(dv.id)}">${escapeText(dv.label)}</text>` +
      `\n  </g>`,
    );
  }
  return groups.join('\n');
}
```
