```ts
import QRCode from 'qrcode';

export interface QrMatrix {
  /** Width/height of the QR symbol in modules (always odd). */
  size: number;
  /** Row-major bit data; data[r*size+c] is 1 (dark) or 0 (light). */
  data: Uint8Array;
}

/** Minimal jspdf surface drawQr needs — keeps it pure & stub-testable. */
export interface QrPdf {
  setFillColor(r: number, g: number, b: number): void;
  rect(x: number, y: number, w: number, h: number, style: string): void;
}

/** Build the QR bit matrix for `text` (synchronous, deterministic). */
export function qrMatrix(text: string): QrMatrix {
  const qr = QRCode.create(text, { errorCorrectionLevel: 'M' });
  return { size: qr.modules.size, data: qr.modules.data };
}

/** Draw the matrix as filled mm-rects starting at (x,y); total side = sizeMm. */
export function drawQr(pdf: QrPdf, m: QrMatrix, x: number, y: number, sizeMm: number): void {
  const moduleMm = sizeMm / m.size;
  pdf.setFillColor(0, 0, 0);
  for (let r = 0; r < m.size; r++) {
    for (let c = 0; c < m.size; c++) {
      if (m.data[r * m.size + c] === 1) {
        pdf.rect(x + c * moduleMm, y + r * moduleMm, moduleMm, moduleMm, 'F');
      }
    }
  }
}
```

```ts
import type { CutListResult, GrainDirection } from '@zilventor/shared';

export interface PanelLabel {
  itemNumber: number;
  partName: string;
  materialName: string;
  length: number;
  width: number;
  thickness: number;
  grain: GrainDirection;
  /** Which edges carry banding (true = banded). */
  edges: { front: boolean; back: boolean; left: boolean; right: boolean };
  copyIndex: number; // 1-based
  copyCount: number; // === row.qty
  qrPayload: string;
}

/**
 * Flatten a cut-list into one label per physical panel: each row → qty labels,
 * copies 1..qty, in cut-list order.
 */
export function buildPanelLabels(result: CutListResult): PanelLabel[] {
  const out: PanelLabel[] = [];
  for (const row of result.rows) {
    for (let k = 1; k <= row.qty; k++) {
      out.push({
        itemNumber: row.itemNumber,
        partName: row.partName,
        materialName: row.materialName,
        length: row.length,
        width: row.width,
        thickness: row.thickness,
        grain: row.grain,
        edges: {
          front: row.edgeBands.front !== null,
          back: row.edgeBands.back !== null,
          left: row.edgeBands.left !== null,
          right: row.edgeBands.right !== null,
        },
        copyIndex: k,
        copyCount: row.qty,
        qrPayload: `Z|${row.itemNumber}|${k}/${row.qty}|${row.length}x${row.width}x${row.thickness}|${row.materialName}`,
      });
    }
  }
  return out;
}
```

```ts
export const LABEL_LAYOUT = {
  cols: 2,
  rows: 5,
  marginMm: 10,
  gutterMm: 4,
  pageW: 210, // A4 portrait
  pageH: 297,
} as const;

export type LabelLayout = typeof LABEL_LAYOUT;

export interface CellRect {
  x: number;
  y: number;
  w: number;
  h: number;
  page: number; // 0-based
}

/** index → label-cell geometry (mm) and 0-based page. Pure pagination math. */
export function labelCellRect(index: number, layout: LabelLayout = LABEL_LAYOUT): CellRect {
  const { cols, rows, marginMm, gutterMm, pageW, pageH } = layout;
  const perPage = cols * rows;
  const cellW = (pageW - 2 * marginMm - (cols - 1) * gutterMm) / cols;
  const cellH = (pageH - 2 * marginMm - (rows - 1) * gutterMm) / rows;
  const page = Math.floor(index / perPage);
  const within = index % perPage;
  const row = Math.floor(within / cols);
  const col = within % cols;
  return {
    x: marginMm + col * (cellW + gutterMm),
    y: marginMm + row * (cellH + gutterMm),
    w: cellW,
    h: cellH,
    page,
  };
}
```

```ts
  right: 'right',
};

export const EDGE_LETTER: Record<PanelEdgeKey, string> = {
  front: 'F',
  back: 'B',
  left: 'L',
  right: 'R',
};

interface Box {
  x: number;
  y: number;
  w: number;
  h: number;
}

/** [x1,y1,x2,y2] for the given side of box b (PDF coords, y grows downward → top = min y). */
export function sideSegment(side: DiagramSide, b: Box): [number, number, number, number] {
  switch (side) {
    case 'top':
      return [b.x, b.y, b.x + b.w, b.y];
    case 'bottom':
      return [b.x, b.y + b.h, b.x + b.w, b.y + b.h];
    case 'left':
      return [b.x, b.y, b.x, b.y + b.h];
    case 'right':
      return [b.x + b.w, b.y, b.x + b.w, b.y + b.h];
  }
}

/** Centered grain arrow segment, or null for grain 'none'. length→horizontal, width→vertical. */
export function grainArrow(grain: GrainDirection, b: Box): [number, number, number, number] | null {
  const cx = b.x + b.w / 2;
  const cy = b.y + b.h / 2;
  if (grain === 'length') return [b.x + b.w * 0.2, cy, b.x + b.w * 0.8, cy];
  if (grain === 'width') return [cx, b.y + b.h * 0.2, cx, b.y + b.h * 0.8];
  return null;
}

export interface DiagramPdf {
  setLineWidth(w: number): void;
  line(x1: number, y1: number, x2: number, y2: number): void;
  rect(x: number, y: number, w: number, h: number, style?: string): void;
  setFontSize(s: number): void;
  text(t: string, x: number, y: number): void;
}

const EDGE_KEYS: PanelEdgeKey[] = ['front', 'back', 'left', 'right'];

/** Draw the mini panel diagram (panel box scaled to length:width, grain arrow, banded edges). */
export function drawLabelDiagram(pdf: DiagramPdf, label: PanelLabel, rect: Box): void {
  // Fit a length:width box inside rect, preserving aspect, with a small inset.
  const inset = 1.5;
  const availW = rect.w - 2 * inset;
  const availH = rect.h - 2 * inset;
  const scale = Math.min(availW / label.length, availH / label.width);
  const pw = label.length * scale;
  const ph = label.width * scale;
  const box: Box = {
    x: rect.x + (rect.w - pw) / 2,
    y: rect.y + (rect.h - ph) / 2,
    w: pw,
    h: ph,
  };

  // Panel outline (thin).
  pdf.setLineWidth(0.2);
  pdf.rect(box.x, box.y, box.w, box.h);

  // Grain arrow.
  const arrow = grainArrow(label.grain, box);
  if (arrow) {
    pdf.setLineWidth(0.3);
    pdf.line(arrow[0], arrow[1], arrow[2], arrow[3]);
  }

  // Banded edges (thick) + letter.
  pdf.setLineWidth(0.9);
  pdf.setFontSize(5);
  for (const key of EDGE_KEYS) {
    if (!label.edges[key]) continue;
    const side = EDGE_SIDE[key];
    const seg = sideSegment(side, box);
    pdf.line(seg[0], seg[1], seg[2], seg[3]);
    // Letter just inside the midpoint of the segment.
    const mx = (seg[0] + seg[2]) / 2;
    const my = (seg[1] + seg[3]) / 2;
    pdf.text(EDGE_LETTER[key], mx, my);
  }
}
```

```ts
/** Draw one label into its cell: border, text block, mini diagram, QR. */
function drawLabelCell(pdf: jsPDF, label: PanelLabel, cell: CellRect): void {
  const pad = 3;
  // Cell border.
  pdf.setDrawColor(0, 0, 0);
  pdf.setLineWidth(0.2);
  pdf.rect(cell.x, cell.y, cell.w, cell.h);

  // Text block (top-left).
  const tx = cell.x + pad;
  let ty = cell.y + pad + 3;
  pdf.setFontSize(9);
  pdf.text(`#${label.itemNumber}  ${label.partName}`, tx, ty);
  ty += 5;
  pdf.setFontSize(7);
  pdf.text(label.materialName, tx, ty);
  ty += 4;
  pdf.text(`${label.length} x ${label.width} x ${label.thickness} mm`, tx, ty);
  ty += 4;
  pdf.text(`copy ${label.copyIndex}/${label.copyCount}`, tx, ty);

  // QR (top-right corner).
  const qrSize = Math.min(cell.h - 2 * pad, cell.w * 0.3);
  drawQr(pdf, qrMatrix(label.qrPayload), cell.x + cell.w - pad - qrSize, cell.y + pad, qrSize);

  // Mini panel diagram (bottom band of the cell).
  const diagTop = ty + 2;
  drawLabelDiagram(pdf, label, {
    x: tx,
    y: diagTop,
    w: cell.w - 2 * pad,
    h: cell.y + cell.h - pad - diagTop,
  });
}

/** Render all labels onto an A4 portrait PDF; returns the file bytes. */
export function renderLabelsPdf(labels: PanelLabel[], layout: LabelLayout = LABEL_LAYOUT): Uint8Array {
  const pdf = new jsPDF({ unit: 'mm', format: 'a4', orientation: 'portrait' });
  const perPage = layout.cols * layout.rows;
  labels.forEach((label, i) => {
    if (i > 0 && i % perPage === 0) pdf.addPage();
    drawLabelCell(pdf, label, labelCellRect(i, layout));
  });
  return new Uint8Array(pdf.output('arraybuffer'));
}
```

Note: `jsPDF` satisfies the `DiagramPdf` and `QrPdf` structural interfaces (it has `setLineWidth`, `line`, `rect`, `setFontSize`, `text`, `setFillColor`), so passing `pdf` directly to `drawLabelDiagram`/`drawQr` typechecks.

```ts
import type { CutListResult } from '@zilventor/shared';
import { buildPanelLabels } from '../drawing/buildPanelLabels';
import { renderLabelsPdf } from '../drawing/renderLabelsPdf';
import { triggerDownload } from '../nesting/triggerDownload';

/** Export one printable PDF label per physical panel from the active assembly cut-list. */
export class ExportPanelLabelsCommand {
  constructor(private opts: { cutListResult: CutListResult; suggestedName: string }) {}

  async apply(): Promise<void> {
    const labels = buildPanelLabels(this.opts.cutListResult);
    const bytes = renderLabelsPdf(labels);
    (globalThis as { __zilventorPanelLabelsBytes?: Uint8Array }).__zilventorPanelLabelsBytes = bytes;
    const blob = new Blob([bytes as unknown as BlobPart], { type: 'application/pdf' });
    await triggerDownload(blob, {
      suggestedName: this.opts.suggestedName,
      mimeType: 'application/pdf',
      extension: 'pdf',
      description: 'Labels PDF',
    });
  }
}
```
