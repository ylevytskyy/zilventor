# M2.1 Per-Panel Manufacturing DXF — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Export a machining-ready DXF per unique panel (outline, layered bores by face/diameter, grooves, edge-bands, grain arrow, info block, native L/W dimensions), bundled one-file-per-machining-signature into a zip, from a Furniture-tab button.

**Architecture:** Three reused seams: (1) a NEW pure builder `renderPanelManufacturingDxf(input)` using the existing `createDxfDocument()`; (2) a NEW gather seam `aggregatePanelManufacturing(...)` that combines the existing `aggregateCutListRecursive` + `aggregateBoresRecursive` walkers + groove extraction (no new tree-walk) and dedups by machining signature; (3) a NEW `ExportPanelManufacturingDxfCommand` that gathers → builds each → `buildDxfZip` → downloads, wired to a new `FurnitureTab` button with en/uk i18n.

**Tech Stack:** TypeScript, pnpm workspaces, Vitest, React (shell), fflate (zip), hand-rolled DXF writer.

**Spec:** `docs/superpowers/specs/2026-06-23-m2.1-panel-manufacturing-dxf-design.md` (design approved; geometry derived from `face2DFrame`).

## Global Constraints

- Coordinate frame: **L→+X, W→+Y, origin bottom-left, mm, 1:1**. NO Y-flip (unlike the drawing DXF builder).
- Layer names live ONLY in the `MANUFACTURING_LAYERS` config object — never inline a layer string elsewhere.
- Back-face (`Bottom`) bore mirror is `(u, v) → (u, W − v)` (Y-axis mirror). NOT `X = L − u`.
- `DxfDocument` method arg order is fixed (see signatures in each task); layer string is always the first arg.
- Bore depth is a TEXT label on `BORE-INFO`, never a layer-name suffix.
- File-uniqueness key = machining signature (geometry L×W×T×material×grain + sorted bores + grooves), NOT the cut-list geometry-only key. `itemNumber` reused from the cut list. Filenames `item-<n>-<partName>.dxf` (sanitize partName: replace `/[^A-Za-z0-9_-]/g` with `_`).
- Tests: app package `cd packages/app && pnpm vitest run <file>`; shared package `cd packages/shared && pnpm vitest run <file>`. Gate: `pnpm typecheck && pnpm lint` at repo root.
- Branch: `feat/m2-manufacturing-dxf` (already created). Commit per task.

```ts
// packages/app/src/drawing/renderPanelManufacturingDxf.ts
import { createDxfDocument, type DxfDocument } from './dxfWriter';
import { MANUFACTURING_LAYERS as L } from './manufacturingLayers';
import type { PanelSpec } from '@zilventor/shared/panel-feature'; // match existing import alias; see note
import type { BoreOp } from '@zilventor/shared/bore';
import type { GrooveFeature } from '@zilventor/shared/groove';

export type PanelManufacturingInput = {
  itemNumber: number;
  partName: string;
  spec: PanelSpec;
  materialName: string;
  bores: BoreOp[];
  grooves: GrooveFeature[];
  qty: number;
};

function drawOutline(doc: DxfDocument, len: number, wid: number): void {
  doc.lwpolyline(L.outline,
    [{ x: 0, y: 0 }, { x: len, y: 0 }, { x: len, y: wid }, { x: 0, y: wid }], true);
}

function drawInfo(doc: DxfDocument, input: PanelManufacturingInput, wid: number): void {
  const { spec } = input;
  const lines = [
    `#${input.itemNumber} ${input.partName}`,
    input.materialName,
    `${spec.length}x${spec.width}x${spec.thickness}`,
    `Qty: ${input.qty}`,
  ];
  const h = 8;
  lines.forEach((s, i) => doc.text(L.info, 0, wid + 20 + (lines.length - 1 - i) * (h + 4), h, s));
}

function drawDim(
  doc: DxfDocument, layer: string, p1: [number, number], p2: [number, number],
  dimLine: [number, number], rotation: number, label: string,
): void {
  const mid: [number, number] = [(p1[0] + p2[0]) / 2, (p1[1] + p2[1]) / 2];
  const textMid: [number, number] = [(dimLine[0] + mid[0]) / 2 + 1, (dimLine[1] + mid[1]) / 2 + 1];
  const blockName = doc.block(() => {
    doc.line(layer, dimLine[0] === p1[0] ? dimLine[0] : p1[0], dimLine[1] === p1[1] ? dimLine[1] : p1[1],
      dimLine[0] === p2[0] ? dimLine[0] : p2[0], dimLine[1] === p2[1] ? dimLine[1] : p2[1]);
    doc.text(layer, textMid[0], textMid[1], 5, label);
  });
  doc.dimension({
    layer, blockName, dimStyle: 'STANDARD', dimType: 0,
    defPoint: dimLine, textMid, text: label, rotation,
    extraPoints: [[13, p1[0], p1[1]], [14, p2[0], p2[1]]],
  });
}

export function renderPanelManufacturingDxf(input: PanelManufacturingInput): string {
  const doc = createDxfDocument();
  const len = input.spec.length;
  const wid = input.spec.width;
  drawOutline(doc, len, wid);
  // L dimension (horizontal, below the panel)
  drawDim(doc, L.dimensions, [0, -15], [len, -15], [0, -15], 0, `${len}`);
  // W dimension (vertical, left of the panel)
  drawDim(doc, L.dimensions, [-15, 0], [-15, wid], [-15, 0], 90, `${wid}`);
  drawInfo(doc, input, wid);
  return doc.toString();
}
```

```ts
// add to renderPanelManufacturingDxf.ts

function boreLabel(b: BoreOp): string {
  return b.depth === 'throughAll' ? `O${b.diameter} THRU` : `O${b.diameter}x${b.depth}`;
}

// returns [flatX, flatY, layer] for a bore, given panel L/W
function boreToFlat(b: BoreOp, len: number, wid: number): { x: number; y: number; layer: string } {
  switch (b.face) {
    case 'Top':    return { x: b.u, y: b.v, layer: L.boresFront(b.diameter) };
    case 'Bottom': return { x: b.u, y: wid - b.v, layer: L.boresBack(b.diameter) };
    case 'Front':  return { x: b.v, y: wid, layer: L.boresEdge('Front', b.diameter) };
    case 'Back':   return { x: len - b.v, y: 0, layer: L.boresEdge('Back', b.diameter) };
    case 'Left':   return { x: 0, y: b.v, layer: L.boresEdge('Left', b.diameter) };
    case 'Right':  return { x: len, y: wid - b.v, layer: L.boresEdge('Right', b.diameter) };
  }
}

function drawBores(doc: DxfDocument, bores: BoreOp[], len: number, wid: number): void {
  for (const b of bores) {
    const f = boreToFlat(b, len, wid);
    doc.circle(f.layer, f.x, f.y, b.diameter / 2);
    doc.text(L.boreInfo, f.x + b.diameter / 2 + 1, f.y, 4, boreLabel(b));
  }
}
```

```ts
// add to renderPanelManufacturingDxf.ts
import type { PanelEdgeKey } from '@zilventor/shared/panel-feature';

function grooveRect(g: GrooveFeature, len: number, wid: number):
  Array<{ x: number; y: number }> | null {
  // Only Top/Bottom grooves are representable in the flat face view.
  if (g.faceRole !== 'Top' && g.faceRole !== 'Bottom') return null;
  const mirror = g.faceRole === 'Bottom';
  if (g.axis === 'u') { // runs along +X at cross-offset = position (in Y)
    let y0 = g.position, y1 = g.position + g.width;
    if (mirror) { y0 = wid - g.position; y1 = wid - (g.position + g.width); }
    return [{ x: 0, y: y0 }, { x: len, y: y0 }, { x: len, y: y1 }, { x: 0, y: y1 }];
  } else { // axis 'v' runs along +Y at cross-offset = position (in X)
    const x0 = g.position, x1 = g.position + g.width;
    return [{ x: x0, y: 0 }, { x: x1, y: 0 }, { x: x1, y: wid }, { x: x0, y: wid }];
  }
}

function drawGrooves(doc: DxfDocument, grooves: GrooveFeature[], len: number, wid: number): void {
  for (const g of grooves) {
    const r = grooveRect(g, len, wid);
    if (r) doc.lwpolyline(L.grooves, r, true);
  }
}

const EDGE_LINE: Record<PanelEdgeKey, (len: number, wid: number) => [number, number, number, number]> = {
  front: (len, wid) => [0, wid, len, wid], // top edge
  back: (len) => [0, 0, len, 0],           // bottom edge
  left: (_len, wid) => [0, 0, 0, wid],     // left edge
  right: (len, wid) => [len, 0, len, wid], // right edge
};

function drawEdgeBands(doc: DxfDocument, spec: PanelSpec, len: number, wid: number): void {
  (Object.keys(EDGE_LINE) as PanelEdgeKey[]).forEach((edge) => {
    if (spec.edgeBands[edge]) {
      const [x1, y1, x2, y2] = EDGE_LINE[edge](len, wid);
      doc.line(L.edgeband, x1, y1, x2, y2);
    }
  });
}

function drawGrain(doc: DxfDocument, spec: PanelSpec, len: number, wid: number): void {
  if (spec.grain === 'none') return;
  const cx = len / 2, cy = wid / 2;
  if (spec.grain === 'length') {
    const a = len * 0.3;
    doc.line(L.grain, cx - a, cy, cx + a, cy);
    doc.line(L.grain, cx + a, cy, cx + a - 10, cy + 6);
    doc.line(L.grain, cx + a, cy, cx + a - 10, cy - 6);
  } else { // 'width'
    const a = wid * 0.3;
    doc.line(L.grain, cx, cy - a, cx, cy + a);
    doc.line(L.grain, cx, cy + a, cx + 6, cy + a - 10);
    doc.line(L.grain, cx, cy + a, cx - 6, cy + a - 10);
  }
}
```

```ts
// packages/app/src/assembly/walkers/aggregatePanelManufacturing.ts
import type { CutListResult, CutListRow } from '@zilventor/shared/cutListRow';
import type { BoringAggregateResult } from './aggregateBoresRecursive';
import type { BoreOp } from '@zilventor/shared/bore';
import type { GrooveFeature } from '@zilventor/shared/groove';
import type { PanelManufacturingInput } from '../../drawing/renderPanelManufacturingDxf';
import type { PanelSpec } from '@zilventor/shared/panel-feature';

function sortBores(bores: BoreOp[]): BoreOp[] {
  return [...bores].sort((a, b) =>
    a.face.localeCompare(b.face) || a.u - b.u || a.v - b.v ||
    a.diameter - b.diameter || String(a.depth).localeCompare(String(b.depth)));
}

export function machiningSignature(
  row: CutListRow, bores: BoreOp[], grooves: GrooveFeature[],
): string {
  const geo = `${row.length}x${row.width}x${row.thickness}|${row.materialId}|${row.grain}`;
  const b = sortBores(bores)
    .map((x) => `${x.face}:${x.u}:${x.v}:${x.diameter}:${x.depth}`).join(',');
  const g = [...grooves]
    .map((x) => `${x.faceRole}:${x.axis}:${x.position}:${x.width}:${x.depth}`)
    .sort().join(',');
  return `${geo}#${b}#${g}`;
}

export function aggregatePanelManufacturing(
  cutList: CutListResult,
  boring: BoringAggregateResult,
  groovesByPart: Map<string, GrooveFeature[]>,
): PanelManufacturingInput[] {
  const boresByPart = new Map(boring.panels.map((p) => [p.partPath, p.bores]));
  const bySig = new Map<string, PanelManufacturingInput>();
  for (const row of cutList.rows) {
    const bores = boresByPart.get(row.partPath) ?? [];
    const grooves = groovesByPart.get(row.partPath) ?? [];
    const sig = machiningSignature(row, bores, grooves);
    const existing = bySig.get(sig);
    if (existing) { existing.qty += row.qty; continue; }
    const spec: PanelSpec = {
      length: row.length, width: row.width, thickness: row.thickness,
      materialId: row.materialId, grain: row.grain, edgeBands: row.edgeBands as any,
    };
    bySig.set(sig, {
      itemNumber: row.itemNumber, partName: row.partName, spec,
      materialName: row.materialName, bores, grooves, qty: row.qty,
    });
  }
  return [...bySig.values()];
}
```

NOTE: `row.edgeBands` is a `CutListEdgeBandCell` map, while `PanelSpec.edgeBands` is `Record<PanelEdgeKey, EdgeBand|null>`. The builder only checks truthiness of each edge (Task 3 `drawEdgeBands`), so the cast is safe for THIS use. If `tsc` rejects the cast, map each cell to `cell ? { materialId, thickness } : null` — inspect `CutListEdgeBandCell` in `packages/shared/src/cutListRow.ts` and convert explicitly rather than `as any`.

```ts
// packages/app/src/commands/ExportPanelManufacturingDxfCommand.ts
import { renderPanelManufacturingDxf, type PanelManufacturingInput } from '../drawing/renderPanelManufacturingDxf';
import { buildDxfZip } from '../drawing/dxfZip';

function sanitize(name: string): string {
  return name.replace(/[^A-Za-z0-9_-]/g, '_');
}

export function buildManufacturingZipEntries(
  inputs: PanelManufacturingInput[],
): Record<string, string> {
  const entries: Record<string, string> = {};
  for (const input of inputs) {
    entries[`item-${input.itemNumber}-${sanitize(input.partName)}.dxf`] =
      renderPanelManufacturingDxf(input);
  }
  return entries;
}

export class ExportPanelManufacturingDxfCommand {
  constructor(private opts: { inputs: PanelManufacturingInput[]; suggestedName: string }) {}
  async apply(): Promise<void> {
    const bytes = buildDxfZip(buildManufacturingZipEntries(this.opts.inputs));
    // Mirror ExportBoringDxfCommand's download path: try showSaveFilePicker, else anchor fallback.
    this.download(bytes, this.opts.suggestedName);
  }
  private download(bytes: Uint8Array, name: string): void {
    const blob = new Blob([bytes], { type: 'application/zip' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url; a.download = name; document.body.appendChild(a); a.click();
    a.remove(); URL.revokeObjectURL(url);
  }
}
```
